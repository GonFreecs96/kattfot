# Dependencies for the governed query helper (assay_query.py).
#
# Only `pyyaml` is mandatory. That is the whole point of the sqlite adapter being
# the default engine: reading the committed fixtures needs nothing beyond the
# standard library plus a YAML parser, so `git clone && pytest && demo` works on a
# fresh machine with no cloud account and no compiled dependencies.
#
# `google-cloud-bigquery` is needed only for ASSAY_ENGINE=bigquery. It is listed
# here rather than in a separate extras file because the helper's own error
# message on the BigQuery path points at this file, and a missing dependency there
# is a hard error (exit 3) — never a silent fall back to fixture numbers.
#
# Install into a venv so you don't fight your system Python (PEP 668):
#   python3 -m venv .venv && .venv/bin/pip install -r requirements.txt
pyyaml>=6.0.3                  # metric frontmatter + funnel-spec parsing
google-cloud-bigquery>=3.41.0  # ASSAY_ENGINE=bigquery only; queries run as you via ADC
